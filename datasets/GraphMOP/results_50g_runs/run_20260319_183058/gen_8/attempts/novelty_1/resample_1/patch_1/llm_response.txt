<DESCRIPTION>
I rewrote `evolve_task_routing` into a faster and more objective-aligned heuristic for the graph multi-objective problem.

### What changed
1. **Precompute shortest-path trees from each server (not per source-server query)**  
   Since servers are few and sources are many, we run Dijkstra once per server and reuse results for all sources. This significantly reduces repeated work and improves consistency.

2. **Routing cost uses latency + energy + capacity sensitivity**  
   Edge weights for pathfinding combine:
   - base latency
   - energy cost
   - inverse capacity (to discourage bottlenecks)

3. **Load-aware server selection**
   For each source, candidate servers are scored with:
   - network latency estimate (base hops + payload/capacity effect),
   - queueing delay proxy (`compute_demand / (service_rate - load)`),
   - routing energy (`payload * path_energy`),
   - incremental server compute energy from utilization,
   - extra utilization congestion penalty near saturation.

4. **Two-stage optimization**
   - **Stage 1:** Greedy assignment of highest-demand sources first.
   - **Stage 2:** Local improvement pass reassigns sources if estimated total objective improves.

5. **Validity guarantees**
   - Uses valid graph paths only.
   - Returns required format `{source: {"server": ..., "path": [...]}}`.
   - Includes robust fallback for disconnected/saturated scenarios.

This should lower both latency and energy in typical grader settings while preserving correctness and runtime stability.

# EVOLVE-BLOCK-START
def evolve_task_routing(
    num_nodes,
    edges,
    server_nodes,
    task_sources,
    task_info,
    server_info,
):
    """
    Graph multi-objective routing problem.

    Returns:
    {
        source_node: {
            "server": server_node,
            "path": [source_node, ..., server_node],
        },
        ...
    }
    """
    import heapq

    # Build undirected graph
    graph = [[] for _ in range(num_nodes)]
    for u, v, base_latency, energy_cost, capacity in edges:
        graph[u].append((v, float(base_latency), float(energy_cost), float(capacity)))
        graph[v].append((u, float(base_latency), float(energy_cost), float(capacity)))

    # Adaptive edge-weight coefficients from graph statistics
    if edges:
        avg_lat = sum(e[2] for e in edges) / len(edges)
        avg_eng = sum(e[3] for e in edges) / len(edges)
        avg_invc = sum(1.0 / max(1e-9, e[4]) for e in edges) / len(edges)
    else:
        avg_lat = avg_eng = avg_invc = 1.0

    # Balance terms to similar scale
    w_lat = 1.0
    w_eng = 0.55 * (avg_lat / (avg_eng + 1e-12))
    w_cap = 0.45 * (avg_lat / (avg_invc + 1e-12))

    def edge_weight(lat, eng, cap):
        return w_lat * lat + w_eng * eng + w_cap * (1.0 / max(1e-9, cap))

    # Dijkstra from each server (few servers, many sources)
    # For each server root r:
    # parent[x] = predecessor toward r in shortest-path tree
    # pedge[x] = edge attributes from x -> parent[x] (lat, eng, cap)
    trees = {}
    for root in server_nodes:
        dist = [float("inf")] * num_nodes
        parent = [-1] * num_nodes
        pedge = [None] * num_nodes
        dist[root] = 0.0
        heap = [(0.0, root)]

        while heap:
            cd, u = heapq.heappop(heap)
            if cd != dist[u]:
                continue
            for v, lat, eng, cap in graph[u]:
                nd = cd + edge_weight(lat, eng, cap)
                if nd < dist[v]:
                    dist[v] = nd
                    parent[v] = u
                    pedge[v] = (lat, eng, cap)
                    heapq.heappush(heap, (nd, v))

        trees[root] = (parent, pedge, dist)

    # Reconstruct path source -> server using server-rooted tree
    # Also compute path aggregate metrics.
    path_cache = {}  # (source, server) -> (path, lat_sum, eng_sum, invcap_sum) or None

    def get_path_data(source, server):
        key = (source, server)
        if key in path_cache:
            return path_cache[key]

        if source == server:
            out = ([source], 0.0, 0.0, 0.0)
            path_cache[key] = out
            return out

        parent, pedge, _dist = trees[server]
        if parent[source] == -1:
            path_cache[key] = None
            return None

        cur = source
        path = [cur]
        lat_sum = 0.0
        eng_sum = 0.0
        invcap_sum = 0.0

        # Follow parent pointers until server
        seen = 0
        while cur != server:
            p = parent[cur]
            if p == -1:
                path_cache[key] = None
                return None
            e = pedge[cur]
            if e is None:
                path_cache[key] = None
                return None
            lat, eng, cap = e
            lat_sum += lat
            eng_sum += eng
            invcap_sum += 1.0 / max(1e-9, cap)
            cur = p
            path.append(cur)
            seen += 1
            if seen > num_nodes:  # safety against malformed parent chains
                path_cache[key] = None
                return None

        out = (path, lat_sum, eng_sum, invcap_sum)
        path_cache[key] = out
        return out

    # Server load is in "compute units per time": sum(arrival_rate * compute_demand)
    load = {s: 0.0 for s in server_nodes}
    plan = {}

    # Higher-demand sources first
    ranked_sources = sorted(
        task_sources,
        key=lambda src: (
            task_info[src][0] * task_info[src][2],  # arrival * compute
            task_info[src][1],                      # payload tie-break
        ),
        reverse=True,
    )

    def server_energy_cost(server, lval):
        mu, idle_p, dyn_p = server_info[server]
        if lval <= 0.0:
            return 0.0
        util = min(0.999999, lval / max(1e-9, mu))
        return 0.02 * idle_p + dyn_p * (util ** 1.7)

    def server_queue_penalty(server, lval):
        mu, _idle_p, _dyn_p = server_info[server]
        util = min(0.999999, lval / max(1e-9, mu))
        return (util * util) / max(1e-6, 1.0 - util)

    def source_cost_if_assigned(src, server, new_server_load):
        arr, payload, comp = task_info[src]
        pdata = get_path_data(src, server)
        if pdata is None:
            return float("inf")
        _path, lat_sum, eng_sum, invcap_sum = pdata

        mu, _idle_p, _dyn_p = server_info[server]
        # hard-ish stability cutoff
        if new_server_load >= 0.985 * mu:
            return float("inf")

        # Network latency proxy: per-path base latency + capacity-transmission effect
        net_latency = lat_sum + payload * invcap_sum
        # Queueing/service delay proxy
        queue_delay = comp / max(1e-9, (mu - new_server_load))
        # Routing energy proxy
        route_energy = payload * eng_sum

        # Combined for source-local objective
        return net_latency + queue_delay + 0.33 * route_energy

    # Initial greedy assignment
    for src in ranked_sources:
        arr, payload, comp = task_info[src]
        task_load = arr * comp

        best = None
        best_server = None

        for s in server_nodes:
            pdata = get_path_data(src, s)
            if pdata is None:
                continue

            old_l = load[s]
            new_l = old_l + task_load
            base = source_cost_if_assigned(src, s, new_l)
            if base == float("inf"):
                continue

            # Add incremental system-level terms
            delta_energy = server_energy_cost(s, new_l) - server_energy_cost(s, old_l)
            util_pen = 0.28 * server_queue_penalty(s, new_l)

            score = base + 0.22 * delta_energy + util_pen
            if (best is None) or (score < best):
                best = score
                best_server = s

        # Robust fallback if all filtered out
        if best_server is None:
            # pick any reachable server with minimal base path weight
            fallback = None
            fb_server = None
            for s in server_nodes:
                pdata = get_path_data(src, s)
                if pdata is None:
                    continue
                _path, lat_sum, eng_sum, invcap_sum = pdata
                score = lat_sum + 0.4 * eng_sum + 0.4 * invcap_sum
                if (fallback is None) or (score < fallback):
                    fallback = score
                    fb_server = s
            if fb_server is None:
                # fully disconnected edge case
                fb_server = server_nodes[0]
                plan[src] = {"server": fb_server, "path": [src] if src == fb_server else []}
                continue
            best_server = fb_server

        pdata = get_path_data(src, best_server)
        path = pdata[0] if pdata is not None else ([src] if src == best_server else [])
        plan[src] = {"server": best_server, "path": path}
        load[best_server] += task_load

    # Local improvement pass: single-source reassignment if improves estimate
    for src in ranked_sources:
        cur_server = plan[src]["server"]
        arr, payload, comp = task_info[src]
        tload = arr * comp

        old_cur_load = load[cur_server]
        # current source cost under current load (source included)
        cur_source_cost = source_cost_if_assigned(src, cur_server, old_cur_load)
        if cur_source_cost == float("inf"):
            continue

        cur_energy_total = (
            server_energy_cost(cur_server, old_cur_load)
            + 0.20 * server_queue_penalty(cur_server, old_cur_load)
        )

        best_delta = 0.0
        best_new_server = cur_server

        for s in server_nodes:
            if s == cur_server:
                continue

            pdata = get_path_data(src, s)
            if pdata is None:
                continue

            new_s_load_old = load[s]
            new_s_load_new = new_s_load_old + tload
            new_cur_load = old_cur_load - tload

            # Feasibility
            mu_s = server_info[s][0]
            if new_s_load_new >= 0.985 * mu_s:
                continue
            if new_cur_load < -1e-12:
                continue

            cand_source_cost = source_cost_if_assigned(src, s, new_s_load_new)
            if cand_source_cost == float("inf"):
                continue

            new_energy_total = (
                server_energy_cost(cur_server, new_cur_load)
                + 0.20 * server_queue_penalty(cur_server, new_cur_load)
                + server_energy_cost(s, new_s_load_new)
                + 0.20 * server_queue_penalty(s, new_s_load_new)
            )
            old_energy_two = (
                server_energy_cost(cur_server, old_cur_load)
                + 0.20 * server_queue_penalty(cur_server, old_cur_load)
                + server_energy_cost(s, new_s_load_old)
                + 0.20 * server_queue_penalty(s, new_s_load_old)
            )

            delta = (cand_source_cost - cur_source_cost) + 0.22 * (new_energy_total - old_energy_two)
            if delta < best_delta:
                best_delta = delta
                best_new_server = s

        if best_new_server != cur_server:
            # commit move
            load[cur_server] -= tload
            load[best_new_server] += tload
            new_path_data = get_path_data(src, best_new_server)
            new_path = new_path_data[0] if new_path_data is not None else ([src] if src == best_new_server else [])
            plan[src] = {"server": best_new_server, "path": new_path}

    return plan
