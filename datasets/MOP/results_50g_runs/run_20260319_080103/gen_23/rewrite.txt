def evolve_drone_path(start, end, school, base, num_points):
    """
    连续空间多目标折线优化：
    - F1: 控制飞行距离（避免过度弯折）
    - F2: 远离学校噪音区（school）
    - F3: 靠近基站降低信号损失（base）
    返回长度为 num_points 的 Y 坐标列表（X 按起终点均匀分布，且不含端点）。
    """
    import math

    x0, y0 = start
    x1, y1 = end
    xs, ys = school
    xb, yb = base

    if num_points <= 0:
        return []

    # ---------- helpers ----------
    def gaussian_weight(x, center, sigma):
        if sigma <= 1e-9:
            return 0.0
        z = (x - center) / sigma
        return math.exp(-z * z)

    def build_interior_xs():
        # 与原实现一致：使用 (i+1)/(n+1) 作为内点比例
        return [x0 + (x1 - x0) * ((i + 1) / (num_points + 1)) for i in range(num_points)]

    def build_linear_baseline():
        return [y0 + (y1 - y0) * ((i + 1) / (num_points + 1)) for i in range(num_points)]

    def apply_school_repulsion(x_arr, y_arr, amp, sigma):
        # 将轨迹从 school.y 推开；方向由当前相对位置决定
        out = y_arr[:]
        for i, x in enumerate(x_arr):
            w = gaussian_weight(x, xs, sigma)
            direction = 1.0 if out[i] >= ys else -1.0
            out[i] += direction * amp * w
        return out

    def apply_base_attraction(x_arr, y_arr, gain, sigma):
        # 向 base.y 拉拢
        out = y_arr[:]
        for i, x in enumerate(x_arr):
            w = gaussian_weight(x, xb, sigma)
            out[i] += (yb - out[i]) * gain * w
        return out

    def smooth_with_endpoint_awareness(y_arr, rounds=2, lam=0.18):
        # 轻量平滑，保证曲线连续；边界邻居使用真实端点
        out = y_arr[:]
        n = len(out)
        if n <= 1:
            return out
        for _ in range(rounds):
            nxt = out[:]
            for i in range(n):
                yl = y0 if i == 0 else out[i - 1]
                yr = y1 if i == n - 1 else out[i + 1]
                nxt[i] = out[i] + lam * (0.5 * (yl + yr) - out[i])
            out = nxt
        return out

    # ---------- staged optimization ----------
    x_arr = build_interior_xs()
    y_arr = build_linear_baseline()

    x_span = max(abs(x1 - x0), 1e-6)

    # 1) 学校规避：局部高斯斥力（避免靠近 school）
    y_arr = apply_school_repulsion(
        x_arr,
        y_arr,
        amp=12.0,
        sigma=0.16 * x_span
    )

    # 2) 基站靠拢：第一阶段（较宽支撑）
    y_arr = apply_base_attraction(
        x_arr,
        y_arr,
        gain=0.92,
        sigma=0.22 * x_span
    )

    # 3) 基站二次局部恢复：第二阶段（窄核 + 小幅）
    #    重点改善 base 附近 F3，同时尽量不破坏全局形状/F2。
    y_arr = apply_base_attraction(
        x_arr,
        y_arr,
        gain=0.35,
        sigma=0.07 * x_span
    )

    # 4) 轻度平滑，压制尖锐变化（平衡 F1）
    y_arr = smooth_with_endpoint_awareness(y_arr, rounds=2, lam=0.18)

    return [float(v) for v in y_arr]
