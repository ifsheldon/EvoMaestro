<DESCRIPTION>
This rewrite replaces the previous local Gaussian field heuristic with a fundamentally different approach: a monotone cubic Bézier trajectory search.

Instead of directly perturbing every sampled Y-point, the path is represented by a smooth low-dimensional curve with two internal Bézier control points. That makes the trajectory naturally smooth and keeps distance growth under control. A deterministic pattern search then optimizes the two control-point Y values using a surrogate objective that jointly balances:

- short total path length,
- staying away from the school,
- moving closer to the base station, especially in the later portion of the route.

This should perform better because it searches over globally smooth curves rather than many locally perturbed polylines, reducing jaggedness and overfitting while still allowing a deliberate bend away from the school and toward the base.

# EVOLVE-BLOCK-START
def evolve_drone_path(start, end, school, base, num_points):
    """
    Smooth multi-objective drone path generation using a monotone cubic Bezier curve.

    Inputs:
        start, end, school, base: (x, y)
        num_points: number of intermediate y-values to return at uniformly spaced x

    Output:
        list[float] of length num_points
    """
    import math

    x0, y0 = start
    x3, y3 = end
    sx, sy = school
    bx, by = base

    if num_points <= 0:
        return []

    dx_total = x3 - x0
    dy_total = y3 - y0

    # Degenerate vertical case: return linear interpolation in y.
    if abs(dx_total) < 1e-12:
        step = dy_total / (num_points + 1)
        return [float(y0 + (i + 1) * step) for i in range(num_points)]

    # Uniform x samples required by the interface.
    xs = [x0 + (i + 1) * dx_total / (num_points + 1) for i in range(num_points)]

    def clamp(v, lo, hi):
        return lo if v < lo else hi if v > hi else v

    def line_y(x):
        t = (x - x0) / dx_total
        return y0 + t * dy_total

    # Keep Bezier x-control points monotone so x(t) is invertible.
    # Slightly earlier/later controls help create "away then toward" bending.
    x1 = x0 + 0.30 * dx_total
    x2 = x0 + 0.78 * dx_total

    # Reference line values.
    y1_line = line_y(x1)
    y2_line = line_y(x2)
    school_line_y = line_y(sx)
    base_line_y = line_y(bx)

    # Direction to move away from school / toward base.
    away_dir = -1.0 if sy >= school_line_y else 1.0
    toward_dir = 1.0 if by >= base_line_y else -1.0

    # Scene scales.
    span_x = abs(dx_total)
    y_span_ref = max(abs(y3 - y0), abs(sy - school_line_y), abs(by - base_line_y), 12.0)

    # Build smart initial targets for the two internal control points.
    # First control: emphasize school avoidance.
    school_strength = min(24.0, 7.0 + 0.55 * abs(sy - school_line_y) + 0.10 * span_x)
    y1_target = y1_line + away_dir * school_strength

    # Second control: emphasize base attraction.
    base_strength = min(30.0, 8.0 + 0.60 * abs(by - base_line_y) + 0.12 * span_x)
    y2_target = y2_line + toward_dir * base_strength

    # If both objectives agree in direction, allow a stronger coherent bend.
    if away_dir == toward_dir:
        y1_target = y1_line + 1.10 * (y1_target - y1_line)
        y2_target = y2_line + 1.18 * (y2_target - y2_line)

    # Soft bounds to avoid pathological curves.
    refs = [y0, y3, sy, by, y1_target, y2_target, y1_line, y2_line]
    y_lo = min(refs) - (18.0 + 0.45 * y_span_ref)
    y_hi = max(refs) + (18.0 + 0.45 * y_span_ref)

    def bezier_xy(t, cy1, cy2):
        omt = 1.0 - t
        omt2 = omt * omt
        t2 = t * t
        x = omt2 * omt * x0 + 3.0 * omt2 * t * x1 + 3.0 * omt * t2 * x2 + t2 * t * x3
        y = omt2 * omt * y0 + 3.0 * omt2 * t * cy1 + 3.0 * omt * t2 * cy2 + t2 * t * y3
        return x, y

    def bezier_y_at_x(x_target, cy1, cy2):
        # Binary search because x(t) is monotone when control x-values are ordered.
        lo, hi = 0.0, 1.0
        for _ in range(28):
            mid = 0.5 * (lo + hi)
            x_mid, _ = bezier_xy(mid, cy1, cy2)
            if (x_mid < x_target) == (dx_total > 0):
                lo = mid
            else:
                hi = mid
        _, y = bezier_xy(0.5 * (lo + hi), cy1, cy2)
        return y

    def sample_curve(cy1, cy2, n=25):
        pts = []
        for i in range(n + 1):
            t = i / float(n)
            pts.append(bezier_xy(t, cy1, cy2))
        return pts

    def surrogate_score(cy1, cy2):
        # Sampled smooth-curve proxy for the hidden evaluator.
        pts = sample_curve(cy1, cy2, n=26)

        # F1 proxy: path length ratio.
        total_len = 0.0
        for i in range(len(pts) - 1):
            dx = pts[i + 1][0] - pts[i][0]
            dy = pts[i + 1][1] - pts[i][1]
            total_len += math.hypot(dx, dy)
        direct_len = math.hypot(dx_total, dy_total) + 1e-9
        f1 = total_len / direct_len

        # F2 proxy: inverse distance to school, more sensitive near school.
        school_pen = 0.0
        for x, y in pts[1:-1]:
            d = math.hypot(x - sx, y - sy)
            school_pen += 1.0 / (d + 2.0)
        school_pen /= max(1, len(pts) - 2)

        # F3 proxy: average distance to base, slightly weighted later in the path.
        base_pen = 0.0
        for i, (x, y) in enumerate(pts[1:-1], start=1):
            prog = i / float(len(pts) - 1)
            w = 0.75 + 0.90 * prog
            d = math.hypot(x - bx, y - by)
            base_pen += w * d
        base_pen /= max(1, len(pts) - 2)

        # Smoothness/shape control: discourage extreme control excursions.
        ctrl_pen = 0.015 * (abs(cy1 - y1_line) + abs(cy2 - y2_line))
        imbalance_pen = 0.008 * abs((cy2 - y2_line) - 1.15 * (cy1 - y1_line))

        # Encourage the intended qualitative shape:
        # near-school sample should be away from school, near-base sample should move toward base.
        ys_school = bezier_y_at_x(sx, cy1, cy2)
        ys_base = bezier_y_at_x(bx, cy1, cy2)

        school_shape_pen = 0.0
        if away_dir > 0:
            school_shape_pen += max(0.0, sy - ys_school)
        else:
            school_shape_pen += max(0.0, ys_school - sy)

        base_shape_pen = 0.0
        if toward_dir > 0:
            base_shape_pen += max(0.0, by - ys_base)
        else:
            base_shape_pen += max(0.0, ys_base - by)

        # Weighted objective. F2 remains important, but we push harder on F3 than the current code.
        return (
            4.2 * f1
            + 18.5 * school_pen
            + 0.18 * base_pen
            + 0.06 * school_shape_pen
            + 0.04 * base_shape_pen
            + ctrl_pen
            + imbalance_pen
        )

    # Deterministic candidate generation around meaningful values.
    def uniq_sorted(vals):
        out = []
        for v in sorted(vals):
            if not out or abs(v - out[-1]) > 1e-9:
                out.append(v)
        return out

    y1_candidates = uniq_sorted([
        y1_line,
        0.5 * (y1_line + y1_target),
        y1_target,
        y1_line + 0.6 * away_dir * school_strength,
        y1_line + 1.2 * away_dir * school_strength,
        sy + 8.0 * away_dir,
        sy + 16.0 * away_dir,
        by,
    ])
    y2_candidates = uniq_sorted([
        y2_line,
        0.5 * (y2_line + y2_target),
        y2_target,
        y2_line + 0.6 * toward_dir * base_strength,
        y2_line + 1.2 * toward_dir * base_strength,
        by,
        0.5 * (by + y2_line),
        sy,
    ])

    y1_candidates = [clamp(v, y_lo, y_hi) for v in y1_candidates]
    y2_candidates = [clamp(v, y_lo, y_hi) for v in y2_candidates]

    best_cy1 = clamp(y1_target, y_lo, y_hi)
    best_cy2 = clamp(y2_target, y_lo, y_hi)
    best_score = surrogate_score(best_cy1, best_cy2)

    # Coarse grid search.
    for cy1 in y1_candidates:
        for cy2 in y2_candidates:
            s = surrogate_score(cy1, cy2)
            if s < best_score:
                best_score = s
                best_cy1, best_cy2 = cy1, cy2

    # Local pattern search refinement.
    step = max(4.0, 0.18 * y_span_ref)
    for _ in range(6):
        improved = False
        trial_pairs = [
            (best_cy1 + step, best_cy2),
            (best_cy1 - step, best_cy2),
            (best_cy1, best_cy2 + step),
            (best_cy1, best_cy2 - step),
            (best_cy1 + 0.7 * step, best_cy2 + step),
            (best_cy1 - 0.7 * step, best_cy2 - step),
            (best_cy1 + 0.5 * step, best_cy2 - 0.5 * step),
            (best_cy1 - 0.5 * step, best_cy2 + 0.5 * step),
        ]
        for cy1, cy2 in trial_pairs:
            cy1 = clamp(cy1, y_lo, y_hi)
            cy2 = clamp(cy2, y_lo, y_hi)
            s = surrogate_score(cy1, cy2)
            if s < best_score:
                best_score = s
                best_cy1, best_cy2 = cy1, cy2
                improved = True
        if not improved:
            step *= 0.55

    # Convert the optimized Bezier curve to required y-values at uniform x positions.
    ys = [bezier_y_at_x(x, best_cy1, best_cy2) for x in xs]

    # Very light endpoint-aware smoothing to suppress inversion noise from x->t search.
    if len(ys) >= 3:
        smoothed = ys[:]
        for i in range(1, len(ys) - 1):
            smoothed[i] = 0.78 * ys[i] + 0.11 * ys[i - 1] + 0.11 * ys[i + 1]
        ys = smoothed

    return [float(clamp(y, y_lo, y_hi)) for y in ys]
