import math

class AdamOptimizer:
    """Standard Adam optimizer implemented from scratch for stable and fast convergence."""
    def __init__(self, size, lr=0.5, beta1=0.9, beta2=0.99, eps=1e-8):
        self.m = [0.0] * size
        self.v = [0.0] * size
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.t = 0
        
    def step(self, params, grads):
        self.t += 1
        lr_t = self.lr * math.sqrt(1.0 - self.beta2**self.t) / (1.0 - self.beta1**self.t)
        for i in range(len(params)):
            self.m[i] = self.beta1 * self.m[i] + (1.0 - self.beta1) * grads[i]
            # Clip gradients to prevent massive explosions
            g = max(min(grads[i], 1000.0), -1000.0)
            self.v[i] = self.beta2 * self.v[i] + (1.0 - self.beta2) * (g ** 2)
            
            # Apply update
            params[i] -= lr_t * self.m[i] / (math.sqrt(self.v[i]) + self.eps)
