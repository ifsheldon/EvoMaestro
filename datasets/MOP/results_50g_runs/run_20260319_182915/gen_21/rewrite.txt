import math

def evolve_drone_path(start, end, school, base, num_points):
    """
    【无人机连续空间多目标折线优化】
    Uses a multi-phase Adam optimizer to minimize the combined objective of:
    - F1: Path length (Tension)
    - F2: School noise proximity (Repulsion)
    - F3: Base station signal drop (Attraction)
    
    Multi-phase weighting:
    - Early phase: Strong school repulsion to establish safe clearance.
    - Middle phase: Strong base attraction to pull the path closer to the base.
    - Late phase: True objective fine-tuning.
    """
    x_start, y_start = start
    x_end, y_end = end
    sx, sy = school
    bx, by = base
    
    dx = (x_end - x_start) / (num_points + 1)
    xs = [x_start + (i + 1) * dx for i in range(num_points)]
    ys = [y_start + (y_end - y_start) * (i + 1) / (num_points + 1) for i in range(num_points)]
    
    # Adam optimizer parameters
    m = [0.0] * num_points
    v = [0.0] * num_points
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    total_steps = 1200
    
    for step in range(1, total_steps + 1):
        progress = step / total_steps
        
        # Multi-phase weighting schedule
        if progress < 0.3:
            w1, w2, w3 = 1.0, 15.0, 0.2
        elif progress < 0.6:
            t = (progress - 0.3) / 0.3
            w1 = 1.0
            w2 = 15.0 * (1 - t) + 1.0 * t
            w3 = 0.2 * (1 - t) + 5.0 * t
        elif progress < 0.8:
            t = (progress - 0.6) / 0.2
            w1 = 1.0
            w2 = 1.0
            w3 = 5.0 * (1 - t) + 1.0 * t
        else:
            w1, w2, w3 = 1.0, 1.0, 1.0
            
        lr = 0.8 * (1.0 - progress * 0.5) # Decays from 0.8 to 0.4
        
        grads = [0.0] * num_points
        
        f1 = 0.0
        f2 = 0.0
        f3 = 0.0
        
        pts_x = [x_start] + xs + [x_end]
        pts_y = [y_start] + ys + [y_end]
        
        # Calculate approximate objective values for relative gradients
        for i in range(num_points + 1):
            f1 += math.hypot(pts_x[i+1] - pts_x[i], pts_y[i+1] - pts_y[i])
            
        for i in range(num_points):
            x = xs[i]
            y = ys[i]
            dist_sq_s = (x - sx)**2 + (y - sy)**2
            f2 += 1.0 / (dist_sq_s + 1e-5)
            
            dist_sq_b = (x - bx)**2 + (y - by)**2
            f3 += math.sqrt(dist_sq_b + 1e-5)
            
        # Compute gradients
        for i in range(num_points):
            y = ys[i]
            x = xs[i]
            
            prev_y = pts_y[i]
            next_y = pts_y[i+2]
            prev_x = pts_x[i]
            next_x = pts_x[i+2]
            
            d1 = math.hypot(x - prev_x, y - prev_y)
            d2 = math.hypot(next_x - x, next_y - y)
            
            grad_f1 = 0.0
            if d1 > 1e-5: grad_f1 += (y - prev_y) / d1
            if d2 > 1e-5: grad_f1 -= (next_y - y) / d2
            
            dist_sq_s = (x - sx)**2 + (y - sy)**2
            grad_f2 = -2.0 * (y - sy) / ((dist_sq_s + 1e-5)**2)
            
            dist_sq_b = (x - bx)**2 + (y - by)**2
            grad_f3 = (y - by) / math.sqrt(dist_sq_b + 1e-5)
            
            # Relative gradients to minimize w1*log(F1) + w2*log(F2) + w3*log(F3)
            grad = w1 * (grad_f1 / f1) + w2 * (grad_f2 / f2) + w3 * (grad_f3 / f3)
            grads[i] = grad
            
        # Adam update step
        for i in range(num_points):
            m[i] = beta1 * m[i] + (1 - beta1) * grads[i]
            v[i] = beta2 * v[i] + (1 - beta2) * (grads[i] ** 2)
            
            m_hat = m[i] / (1 - beta1 ** step)
            v_hat = v[i] / (1 - beta2 ** step)
            
            ys[i] -= lr * m_hat / (math.sqrt(v_hat) + epsilon)
            
    return ys
# EVOLVE-BLOCK-END
